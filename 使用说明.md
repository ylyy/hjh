# AI助手集成使用说明

## 快速开始

### 1. Electron端配置

#### 安装依赖
```bash
cd AIAssistant
npm install express body-parser cors
```

#### 集成API服务器

在你的 `main.js` 中添加：

```javascript
const { app, BrowserWindow } = require('electron');
const AIApiServer = require('./electron-api-server');

let apiServer;

app.on('ready', async () => {
  // 创建主窗口（你原有的代码）
  createMainWindow();

  // 启动API服务器
  apiServer = new AIApiServer({
    port: 8765,
    apiToken: process.env.API_TOKEN || 'your-secret-token'
  });

  try {
    await apiServer.start();
    console.log('API服务器已启动');
  } catch (error) {
    console.error('API服务器启动失败:', error);
  }
});

app.on('will-quit', async () => {
  // 应用退出时停止API服务器
  if (apiServer) {
    await apiServer.stop();
  }
});
```

#### 接入你的AI服务

修改 `electron-api-server.js` 中的 `callAIService` 方法：

```javascript
async callAIService(prompt, options = {}) {
  // 替换为你的AI服务调用
  // 例如：OpenAI、Claude、本地模型等
  
  const OpenAI = require('openai');
  const openai = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY
  });

  const completion = await openai.chat.completions.create({
    model: "gpt-4",
    messages: [
      {
        role: "system",
        content: "你是一个专业的软件调试助手。"
      },
      {
        role: "user",
        content: prompt
      }
    ],
    temperature: 0.7,
    max_tokens: 2000
  });

  return completion.choices[0].message.content;
}
```

##### 方式一：在现有错误日志界面添加按钮

```csharp
public class YourExistingErrorLogForm : Form
{
    private AIAssistantClient _aiClient;
    
    public YourExistingErrorLogForm()
    {
        InitializeComponent();
        
        // 初始化AI客户端
        _aiClient = new AIAssistantClient(
            "http://localhost:8765",
            "your-secret-token"
        );
        
        // 添加AI分析按钮
        AddAIAnalysisButton();
    }
    
    private void AddAIAnalysisButton()
    {
        var aiButton = new Button
        {
            Text = "🤖 AI分析",
            Location = new Point(10, 10),
            Size = new Size(120, 30)
        };
        aiButton.Click += async (s, e) => await AnalyzeCurrentError();
        
        this.Controls.Add(aiButton);
    }
    
    private async Task AnalyzeCurrentError()
    {
        // 获取当前显示的错误日志
        string errorLog = GetCurrentErrorLog();
        
        // 调用AI分析
        var response = await _aiClient.AnalyzeErrorAsync(errorLog);
        
        if (response.Success)
        {
            // 显示分析结果
            ShowSolution(response.Solution);
        }
        else
        {
            MessageBox.Show($"分析失败: {response.Error}");
        }
    }
    
    private string GetCurrentErrorLog()
    {
        // 返回当前错误日志内容
        // 根据你的实际UI调整
        return errorLogTextBox.Text;
    }
    
    private void ShowSolution(string solution)
    {
        // 显示解决方案
        // 可以弹出新窗口、在侧边栏显示等
        MessageBox.Show(solution, "AI分析结果", 
            MessageBoxButtons.OK, 
            MessageBoxIcon.Information);
    }
}
```

### 3. 配置管理

#### 创建配置类

```csharp
using System;
using System.IO;
using Newtonsoft.Json;

public class AppConfig
{
    public AIServiceConfig AIService { get; set; }
    public ElectronAppConfig ElectronApp { get; set; }
    
    public static AppConfig Load(string configPath = "config.json")
    {
        if (!File.Exists(configPath))
        {
            // 创建默认配置
            var defaultConfig = new AppConfig
            {
                AIService = new AIServiceConfig
                {
                    ApiBaseUrl = "http://localhost:8765",
                    ApiToken = "your-secret-token",
                    Timeout = 30
                },
                ElectronApp = new ElectronAppConfig
                {
                    Path = @"C:\Program Files\AIAssistant\AIAssistant.exe",
                    AutoStart = true
                }
            };
            
            File.WriteAllText(configPath, JsonConvert.SerializeObject(defaultConfig, Formatting.Indented));
            return defaultConfig;
        }
        
        var json = File.ReadAllText(configPath);
        return JsonConvert.DeserializeObject<AppConfig>(json);
    }
}

public class AIServiceConfig
{
    public string ApiBaseUrl { get; set; }
    public string ApiToken { get; set; }
    public int Timeout { get; set; }
}

public class ElectronAppConfig
{
    public string Path { get; set; }
    public bool AutoStart { get; set; }
}
```

## 故障排查

### 常见问题

#### 1. 无法连接到AI服务

**问题**: C#应用提示"无法连接到AI服务"

**解决方案**:
- 检查Electron应用是否正在运行
- 检查端口8765是否被占用
- 检查防火墙是否阻止了连接
- 验证配置文件中的API地址和Token是否正确

```bash
# Windows检查端口占用
netstat -ano | findstr :8765

# 测试API服务
curl http://localhost:8765/api/health
```

#### 2. API Token认证失败

**问题**: 收到401 Unauthorized错误

**解决方案**:
- 确保Electron端和C#端使用相同的API Token
- 检查配置文件中的apiToken设置
- 临时禁用认证进行测试（不推荐生产环境）

#### 3. 请求超时

**问题**: AI分析请求超时

**解决方案**:
- 增加timeout设置（默认30秒）
- 检查AI服务响应时间
- 考虑使用异步回调而不是同步等待
- 对长日志进行截断或摘要

#### 4. Electron应用无法启动

**问题**: AIServiceManager无法启动Electron应用

**解决方案**:
```csharp
// 检查路径是否正确
string appPath = config.ElectronApp.Path;
if (!File.Exists(appPath))
{
    MessageBox.Show($"找不到AI助手应用: {appPath}\n\n请检查配置文件中的路径设置");
}

// 手动启动一次，查看错误信息
Process.Start(appPath);
```

### 2. 缓存机制

```csharp
// 缓存相似错误的分析结果
public class AIAnalysisCache
{
    private Dictionary<string, CacheItem> _cache = new Dictionary<string, CacheItem>();
    
    public string GetCachedSolution(string errorLog)
    {
        string hash = ComputeHash(errorLog);
        
        if (_cache.TryGetValue(hash, out var item))
        {
            if (DateTime.Now - item.Timestamp < TimeSpan.FromHours(24))
            {
                return item.Solution;
            }
        }
        
        return null;
    }
    
    public void CacheSolution(string errorLog, string solution)
    {
        string hash = ComputeHash(errorLog);
        _cache[hash] = new CacheItem
        {
            Solution = solution,
            Timestamp = DateTime.Now
        };
    }
    
    private string ComputeHash(string input)
    {
        using (var sha256 = System.Security.Cryptography.SHA256.Create())
        {
            byte[] bytes = sha256.ComputeHash(Encoding.UTF8.GetBytes(input));
            return BitConverter.ToString(bytes).Replace("-", "");
        }
    }
}
```

### 3. 异步处理

```csharp
// 使用队列处理多个分析请求
public class AIAnalysisQueue
{
    private Queue<AnalysisRequest> _queue = new Queue<AnalysisRequest>();
    private bool _isProcessing = false;
    
    public async Task EnqueueAsync(string errorLog, Action<string> callback)
    {
        _queue.Enqueue(new AnalysisRequest
        {
            ErrorLog = errorLog,
            Callback = callback
        });
        
        if (!_isProcessing)
        {
            await ProcessQueueAsync();
        }
    }
    
    private async Task ProcessQueueAsync()
    {
        _isProcessing = true;
        
        while (_queue.Count > 0)
        {
            var request = _queue.Dequeue();
            var response = await _aiClient.AnalyzeErrorAsync(request.ErrorLog);
            
            if (response.Success)
            {
                request.Callback?.Invoke(response.Solution);
            }
        }
        
        _isProcessing = false;
    }
}
```
